# javac and javap ran on the host (Temurin JDK 25.0.4.1).
javac -g AddWarmup.java
javap -c AddWarmup

# Linux ARM64, Temurin JRE 25.0.5+8, executing the class built above.
docker run --rm --network=none -v "$PWD:/app:ro" -w /app \
  eclipse-temurin@sha256:ecdc0adb37bde1a281c4d245a3cd535004b1fc8322d560f5b09463b248c81df0 \
  java -Xbatch -XX:-TieredCompilation -XX:CompileThreshold=100 \
  -XX:CompileCommand=dontinline,AddWarmup::sum \
  -XX:CompileCommand=print,AddWarmup::sum AddWarmup > jit.txt

# Extract this nmethod's emitted bytes. Relative offsets vary by build.
python3 - <<'PYTHON'
import re
from pathlib import Path
text = Path("jit.txt").read_text()
entry = text.split("[Verified Entry Point]", 1)[1].split("[Exception Handler]", 1)[0]
rows = re.findall(r"^  0x[0-9a-f]+: ([0-9a-f |]+)$", entry, re.M)
Path("sum.bin").write_bytes(bytes.fromhex(" ".join(row.replace("|", " ") for row in rows)))
PYTHON
